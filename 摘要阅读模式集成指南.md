### 步骤2：UI集成

#### 2.1 修改ReadBookActivity
```kotlin
// 在ReadBookActivity.kt中添加
class ReadBookActivity : BaseReadBookActivity() {
    
    private lateinit var readModeManager: ReadModeManager
    private lateinit var modeSwitchButton: ModeSwitchButton
    private lateinit var summaryLoadingView: SummaryLoadingView
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // 初始化模式管理器
        readModeManager = ReadModeManager(this)
        
        // 初始化UI组件
        setupModeSwitchButton()
        setupSummaryLoadingView()
        
        // 监听模式变化
        readModeManager.currentMode.observe(this) { mode ->
            onReadModeChanged(mode)
        }
    }
    
    private fun setupModeSwitchButton() {
        modeSwitchButton = findViewById(R.id.mode_switch_button)
        modeSwitchButton.setCurrentMode(readModeManager.getCurrentMode())
        modeSwitchButton.onModeChangeListener = { newMode ->
            switchReadMode(newMode)
        }
    }
    
    private fun setupSummaryLoadingView() {
        summaryLoadingView = findViewById(R.id.summary_loading_view)
        summaryLoadingView.onRetryClickListener = {
            retrySummaryGeneration()
        }
    }
    
    private fun switchReadMode(newMode: ReadModeManager.ReadMode) {
        val currentChapter = ReadBook.curChapter ?: return
        
        summaryLoadingView.showLoading()
        
        readModeManager.switchMode(newMode, currentChapter) { success ->
            if (success) {
                // 更新显示内容
                loadContent()
                // 更新章节标题
                updateChapterTitle()
            } else {
                summaryLoadingView.showError("摘要生成失败，请重试")
            }
            summaryLoadingView.hide()
        }
    }
    
    private fun updateChapterTitle() {
        val chapterTitle = ReadBook.curChapter?.title ?: ""
        val displayTitle = readModeManager.getChapterTitle(chapterTitle)
        // 更新标题显示
        binding.titleBar.title = displayTitle
    }
}
```
